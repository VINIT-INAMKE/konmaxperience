Receive mutation:
```typescript
const receiveMutation = useMutation({
  mutationFn: () => apiClient.post(`/purchase-orders/${id}/receive`, {
    lines: Object.entries(receivedQuantities)
      .filter(([_, qty]) => Number(qty) > 0)
      .map(([lineId, qty]) => ({
        id: lineId,
        received_quantity: Number(qty),
      })),
  }),
  onSuccess: () => {
    queryClient.invalidateQueries({ queryKey: ['purchase-orders'] });
    queryClient.invalidateQueries({ queryKey: ['inventory'] });
    toast.success('Receiving recorded. Stock levels updated.');
  },
  onError: () => toast.error('Failed to record receiving. Stock levels were not updated. Try again or contact your administrator.'),
});
```
