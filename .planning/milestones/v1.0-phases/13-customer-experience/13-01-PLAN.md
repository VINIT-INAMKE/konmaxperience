model Feedback {
  id             String   @id @default(uuid())
  order_id       String?
  order          Order?   @relation(fields: [order_id], references: [id])
  rating         Int      // 1-5
  comment        String?
  customer_name  String?
  customer_phone String?
  created_at     DateTime @default(now())

model Event {
  id          String         @id @default(uuid())
  title       String
  event_type  String         // "dining" | "workshop" | "pop_up" | "tasting" | "other"
  date        DateTime
  capacity    Int
  price       Decimal
  zone_id     String?
  brand_id    String?
  description String?
  image_url   String?
  status      String         @default("upcoming") // upcoming | past | cancelled
  created_at  DateTime       @default(now())
  updated_at  DateTime       @updatedAt
  zone        Zone?          @relation(fields: [zone_id], references: [id])
  brand       Brand?         @relation(fields: [brand_id], references: [id])
  bookings    EventBooking[]
}

model EventBooking {
  id             String   @id @default(uuid())

  customer_phone String

}

- `rating`: optional int (filter by exact rating)
- `date_from`: optional string (ISO date)
- `date_to`: optional string (ISO date)

`feedback.service.ts`:
- `submit(dto)`: Create feedback record. If `order_id` provided, verify order exists (findUnique, throw NotFoundException if not found). Return created feedback.
- `findAll(filters)`: Query with optional where clauses: `rating` exact match, `created_at >= date_from`, `created_at <= date_to`. Include `{ order: { select: { id: true } } }` per Research Pitfall 6. Order by `created_at` DESC.
- `getStats()`: Use Prisma aggregate to compute average rating and total count. Return `{ average_rating: number, total_count: number }`.

`feedback.controller.ts`:
- `@Post()` + `@Public()`: calls `submit(dto)`. No auth required (per D-01, D-02).
- `@Get()` + `@RequiresPermission(Permission.MANAGE_POS)`: calls `findAll(filters)`.
- `@Get('stats')` + `@RequiresPermission(Permission.MANAGE_POS)`: calls `getStats()`. Place BEFORE any parameterized route to prevent shadowing.

`events.module.ts`: Import PrismaModule, declare EventsController + EventsService.

`dto/create-event.dto.ts`:
- `title`: required string, `@IsString()`, `@MinLength(3)`, `@MaxLength(200)`
- `event_type`: required string, `@IsString()` (dining, workshop, pop_up, tasting, other)
- `date`: required string (ISO), `@IsDateString()`
- `capacity`: required int, `@IsInt()`, `@Min(1)`
- `price`: required number, `@IsNumber()`
- `zone_id`: optional string, `@IsOptional()`, `@IsUUID()`
- `brand_id`: optional string, `@IsOptional()`, `@IsUUID()`
- `description`: optional string, `@IsOptional()`, `@IsString()`, `@MaxLength(2000)`
- `image_url`: optional string, `@IsOptional()`, `@IsString()`

`dto/update-event.dto.ts`: PartialType of CreateEventDto + `status` field (optional string).

`dto/create-booking.dto.ts`:
- `customer_name`: required string, `@IsString()`, `@MinLength(1)`, `@MaxLength(100)`
- `customer_phone`: required string, `@IsString()`, `@MinLength(5)`, `@MaxLength(20)`
- `guests`: required int, `@IsInt()`, `@Min(1)`

`events.service.ts`:
- `findUpcoming()`: Query events where `date >= new Date()` AND `status != 'cancelled'`. Include `bookings` (select only `guests`). Order by `date` ASC. For each event, compute `booked_guests` from sum of booking.guests and `spots_remaining = capacity - booked_guests`.
- `findAll()`: Query all events with bookings count. For ops view.
- `findOne(id)`: findUnique or throw. Include zone, brand, bookings.
- `create(dto)`: Create event. Return with include zone, brand.
- `update(id, dto)`: Update event. Return with include zone, brand.
- `remove(id)`: Delete event (cascades to bookings via schema onDelete).
- `createBooking(eventId, dto)`: Use `this.prisma.$transaction(async (tx) => { ... })` per Research Pattern 2:
  1. `tx.event.findUniqueOrThrow({ where: { id: eventId } })`
  2. `tx.eventBooking.aggregate({ where: { event_id: eventId }, _sum: { guests: true } })`
  3. Check `(booked + dto.guests) > event.capacity` — throw BadRequestException with message: `Sorry, this event is full. No spots remain for ${dto.guests} guests.`
  4. Also check event date is in the future, throw if past.
  5. Create and return booking.
- `getBookings(eventId)`: Query all bookings for an event, ordered by created_at DESC.

`events.controller.ts`:
- `@Get()` + `@Public()`: calls `findUpcoming()`. Public event listing.
- `@Get('all')` + `@RequiresPermission(Permission.MANAGE_OPS)`: calls `findAll()`. Ops view.
- `@Get(':id')` + `@Public()`: calls `findOne(id)`. Public event detail.
- `@Post()` + `@RequiresPermission(Permission.MANAGE_OPS)`: calls `create(dto)`.
- `@Patch(':id')` + `@RequiresPermission(Permission.MANAGE_OPS)`: calls `update(id, dto)`.
- `@Delete(':id')` + `@RequiresPermission(Permission.MANAGE_OPS)`: calls `remove(id)`.
- `@Post(':id/bookings')` + `@Public()`: calls `createBooking(id, dto)`. Public booking submission.
- `@Get(':id/bookings')` + `@RequiresPermission(Permission.MANAGE_OPS)`: calls `getBookings(id)`. Ops booking list.

IMPORTANT: Place `@Get('all')` BEFORE `@Get(':id')` to prevent NestJS route shadowing (same pattern as `daily-summary` before `:id` in OrdersController).

import * as QRCode from 'qrcode';

async generateQr(orderId: string): Promise<{ qr_data_url: string }> {
  // Verify order exists
  await this.prisma.order.findUniqueOrThrow({ where: { id: orderId } });
  const frontendUrl = process.env.FRONTEND_URL || 'http://localhost:3000';
  const url = `${frontendUrl}/feedback/${orderId}`;
  const dataUrl = await QRCode.toDataURL(url, {
    width: 256,
    margin: 2,
    color: { dark: '#000000', light: '#ffffff' },
  });
  return { qr_data_url: dataUrl };
}
