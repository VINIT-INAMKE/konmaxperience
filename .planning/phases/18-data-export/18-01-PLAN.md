   ```typescript
   async putObjectDirect(key: string, body: Buffer, contentType: string): Promise<void> {
     await this.s3.send(new PutObjectCommand({
       Bucket: this.bucketName,
       Key: key,
       Body: body,
       ContentType: contentType,
       ContentLength: body.length,
     }));
   }
   ```

   ```typescript
   import { IsIn, IsOptional, IsString, IsDateString } from 'class-validator';
   import { REPORT_TYPES, ReportType } from '../export-types';

   export class GenerateExportDto {
     @IsIn(REPORT_TYPES)
     reportType: ReportType;

     @IsIn(['csv', 'xlsx'])
     format: 'csv' | 'xlsx';

     @IsOptional()
     @IsDateString()
     dateFrom?: string;

     @IsOptional()

     dateTo?: string;
   ```

   - Implement `buildExportKey(reportType: string, format: string, dateFrom?: string, dateTo?: string): string`
     - Pattern: `exports/{reportType}/{YYYYMMDD}/{reportType}_{dateRange}_{timestamp}.{format}`
     - dateRange: `{dateFrom}_to_{dateTo}` if both provided, else `all`
     - timestamp: `new Date().toISOString().replace(/[-:]/g, '').slice(0, 15)` (e.g., `20260323T120000`)

     ```typescript
     export interface ExportBuilder {

     }
     ```
   - The `generateExport` method checks the builders map, throws if not found, calls `builder.fetchData()`, then calls `builder.buildXlsx()` or `builder.buildCsv()` based on `dto.format`
