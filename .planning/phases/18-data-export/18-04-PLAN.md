   async findAllForExport(dateFrom?: string, dateTo?: string): Promise<WasteLog[]> {
     const where: Record<string, unknown> = {};
     if (dateFrom || dateTo) {
       where.created_at = {};
       if (dateFrom) (where.created_at as any).gte = new Date(dateFrom);
       if (dateTo) (where.created_at as any).lte = new Date(dateTo + 'T23:59:59.999Z');
     }
     return this.prisma.wasteLog.findMany({
