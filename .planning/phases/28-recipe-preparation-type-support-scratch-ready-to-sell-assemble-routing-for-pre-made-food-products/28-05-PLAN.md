    **E) Supply Usage page route:**

    Create `frontend/app/(ops)/operations/kitchen/supply-usage/page.tsx`. Mirror the waste log page pattern exactly:
    ```tsx
    'use client';
    import { useQuery, useQueryClient } from '@tanstack/react-query';
    import { ClipboardList } from 'lucide-react';
    import { apiClient } from '@/lib/api-client';
    import type { SupplyUsageEntry } from '@/lib/types/kitchen';
    import { SupplyUsageForm } from '@/components/ops/kitchen/supply-usage/SupplyUsageForm';
    import { SupplyUsageRow } from '@/components/ops/kitchen/supply-usage/SupplyUsageRow';

    export default function SupplyUsagePage() {
      const queryClient = useQueryClient();
      const { data: entries, isLoading } = useQuery({
        queryKey: ['supply-usage'],
        queryFn: () => apiClient.get<SupplyUsageEntry[]>('/kitchen/supply-usage'),
      });

      const handleSuccess = () => {
        void queryClient.invalidateQueries({ queryKey: ['supply-usage'] });
      };

      return (
        <div className="space-y-8">
          <h1 className="text-2xl font-bold">Supply Usage</h1>
          <div className="grid lg:grid-cols-3 gap-6">
            {/* Left: usage history table */}
            <div className="lg:col-span-2">
              {/* Table or empty state */}
              {isLoading && <div>Loading...</div>}
              {!isLoading && (!entries || entries.length === 0) && (
                <div className="flex flex-col items-center justify-center py-16 text-center">
                  <ClipboardList className="size-12 text-muted-foreground mb-4" />
                  <h3 className="text-lg font-semibold">No usage logged yet</h3>
                  <p className="text-sm text-muted-foreground">Use the form to record supply usage at the end of each shift.</p>
                </div>
              )}
              {entries && entries.length > 0 && (
                <div className="space-y-1">
                  {/* Table header */}
                  <div className="grid grid-cols-6 gap-4 text-xs text-muted-foreground uppercase tracking-wide px-3 py-2">
                    <span>Date</span><span>Supply</span><span>Qty</span><span>Unit</span><span>Notes</span><span>By</span>
                  </div>
                  {entries.map((entry) => (
                    <SupplyUsageRow key={entry.id} entry={entry} />
                  ))}
                </div>
              )}
            </div>
            {/* Right: log form */}
            <div>
              <SupplyUsageForm onSuccess={handleSuccess} />
            </div>
          </div>
        </div>
      );
    }
    ```
