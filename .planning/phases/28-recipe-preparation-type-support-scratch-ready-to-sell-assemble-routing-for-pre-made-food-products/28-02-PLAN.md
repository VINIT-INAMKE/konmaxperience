    Add a new private method that performs FIFO deduction against PrepBatch records for a given recipe_id. Copy the pattern from `deductItemIngredients` lines 712-762 (the FIFO batch deduction section). This method is needed because `deductItemIngredients` operates on BOM lines, but batch_prepared deduction targets the recipe_id directly.

    ```typescript
    private async deductBatchPrepared(
      tx: Prisma.TransactionClient,
      recipeId: string,
      quantity: number,
      userId: string,
      orderId: string,
    ) {
      // FIFO: consume from batches expiring soonest first (per D-10)
      const batches = await tx.prepBatch.findMany({
        where: { recipe_id: recipeId, status: 'active' },
        orderBy: [{ expires_at: 'asc' }, { created_at: 'asc' }],
      });

      let remaining = quantity;
      for (const batch of batches) {
        if (remaining <= 0) break;
        const batchQty = Number(batch.quantity_remaining);
        const deductFromBatch = Math.min(batchQty, remaining);

        await tx.prepBatch.update({
          where: { id: batch.id },
          data: {
            quantity_remaining: { decrement: deductFromBatch },
            status: batchQty - deductFromBatch <= 0 ? 'depleted' : 'active',
          },
        });

        remaining -= deductFromBatch;
      }

      if (remaining > 0) {
        // Insufficient batch quantity — log but don't throw (order already confirmed)
        console.warn(
          `Insufficient batch quantity for recipe ${recipeId}: needed ${quantity}, short by ${remaining}`,
        );
      }
    }
    ```
