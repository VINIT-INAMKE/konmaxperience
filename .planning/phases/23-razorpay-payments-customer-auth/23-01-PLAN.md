    import Redis from 'ioredis';

    @Injectable()
    export class RedisService implements OnModuleInit, OnModuleDestroy {
      private client: Redis | null = null;

      onModuleInit() {
        const url = process.env.UPSTASH_REDIS_URL;
        if (!url) {
          console.warn('[CustomerAuth] UPSTASH_REDIS_URL not set — OTP storage disabled');
          return;
        }
        this.client = new Redis(url, {
          maxRetriesPerRequest: 3,
          enableReadyCheck: false,
          connectTimeout: 5000,
          lazyConnect: true,
        });
        this.client.on('error', (err) => console.error('[Redis]', err.message));
      }

      onModuleDestroy() {
        this.client?.disconnect();
      }

      getClient(): Redis | null {
        return this.client;
      }
    }

    **1. Create WhatsAppService at `backend/src/customer-auth/whatsapp.service.ts` (per D-02):**

    Follow PusherService graceful fallback pattern:
    ```typescript
    import { Injectable } from '@nestjs/common';

    @Injectable()
    export class WhatsAppService {
      private token: string | null;
      private phoneId: string | null;

      constructor() {
        this.token = process.env.WHATSAPP_TOKEN || null;
        this.phoneId = process.env.WHATSAPP_PHONE_ID || null;
      }

      async sendOtp(recipientPhone: string, otp: string): Promise<void> {
        // Dev fallback — log to console if WhatsApp not configured
        if (!this.token || !this.phoneId) {
          console.log(`[DEV] OTP for ${recipientPhone}: ${otp}`);
          return;
        }

        // Normalize phone: ensure 91 prefix, no +
        const normalized = recipientPhone.startsWith('91')
          ? recipientPhone
          : `91${recipientPhone}`;

        const url = `https://graph.facebook.com/v18.0/${this.phoneId}/messages`;
        const res = await fetch(url, {
          method: 'POST',
          headers: {
            Authorization: `Bearer ${this.token}`,
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({
            messaging_product: 'whatsapp',
            to: normalized,
            type: 'template',
            template: {
              name: 'otp_verification',
              language: { code: 'en' },
              components: [
                {
                  type: 'body',
                  parameters: [{ type: 'text', text: otp }],
                },
              ],
            },
          }),
        });

        if (!res.ok) {
          const body = await res.text();
          console.error(`[WhatsApp] Failed to send OTP: ${res.status} ${body}`);
          throw new Error(`WhatsApp API error: ${res.status}`);
        }
      }
    }
    ```

    **2. Create CustomerAuthService at `backend/src/customer-auth/customer-auth.service.ts` (per D-02, D-03, D-04, D-05, D-15):**

    Inject: PrismaService, JwtService, ConfigService, RedisService, WhatsAppService.

    Methods:
    - `sendOtp(phone: string)`:
      - Get Redis client. If null, throw ServiceUnavailableException.
      - Rate limit check: `INCR otp_rate:{phone}`. If first, `EXPIRE 3600`. If > 3, throw TooManyRequestsException('Too many attempts — try again in 1 hour').
      - Generate OTP: `crypto.randomInt(100000, 999999).toString()`
      - Hash: `bcrypt.hash(otp, 10)`
      - Store: `redis.set('otp:' + phone, hash, 'EX', 300)`
      - Send: `whatsappService.sendOtp(phone, otp)`
      - Return: `{ message: 'OTP sent' }`

    - `verifyOtp(phone: string, submittedOtp: string, res: Response)`:
      - Get Redis client. If null, throw ServiceUnavailableException.
      - Fetch stored hash: `redis.get('otp:' + phone)`. If null, throw GoneException('This code has expired — request a new one').
      - Compare: `bcrypt.compare(submittedOtp, storedHash)`. If false, throw UnauthorizedException('Incorrect code — check your WhatsApp and try again').
      - Delete OTP key: `redis.del('otp:' + phone)` (consume — prevent reuse).
      - Upsert Customer: `prisma.customer.upsert({ where: { phone }, create: { phone }, update: {} })`.
      - Check if first login (customer.created_at within last 5 seconds): if so, run auto-link:
        ```typescript
        await prisma.order.updateMany({ where: { customer_phone: phone, customer_id: null }, data: { customer_id: customer.id } });
        await prisma.eventBooking.updateMany({ where: { customer_phone: phone, customer_id: null }, data: { customer_id: customer.id } });
        await prisma.feedback.updateMany({ where: { customer_phone: phone, customer_id: null }, data: { customer_id: customer.id } });
        ```
      - Sign JWT: `jwtService.sign({ customerId: customer.id, type: 'customer' }, { expiresIn: '30d' })`
      - Set cookie: `res.cookie('customer_access_token', token, { httpOnly: true, secure: process.env.NODE_ENV === 'production', sameSite: 'lax', maxAge: 30 * 24 * 60 * 60 * 1000, path: '/' })`
      - Return: `{ customer: { id, phone, name, email }, isNewCustomer: (first login check) }`
