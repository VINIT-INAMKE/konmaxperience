  brand_type    String

  notes         String?

model Channel {
  id           String @id @default(uuid())

  channel_type String

model Asset {
  id              String   @id @default(uuid())
  name            String
  asset_type      String
  linked_task_id  String?

  linked_brand_id String?
  linked_brand    Brand?   @relation(fields: [linked_brand_id], references: [id])
  url             String
  status          String   @default("draft")
