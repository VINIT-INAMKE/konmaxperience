**3. Handle recipe parsing specially in `parseFile`:**
The recipe type uses `parseRecipeXLSX` (imported in Plan 01) which returns `{ headers, bomLines }`. The parseFile method needs special handling for recipes:

After the mimetype check and before `normalizedRows`, add:
```typescript
if (importType === 'recipes') {
  // Recipe uses multi-sheet parser
  const { headers: recipeHeaders, bomLines } = await parseRecipeXLSX(buffer);

  if (recipeHeaders.length === 0) {
    throw new BadRequestException('Sheet 1 (Recipes) contains no data rows');
  }
  if (recipeHeaders.length + bomLines.length > 500) {
    throw new BadRequestException('This file exceeds the 500-row limit. Split it into smaller files.');
  }

  // Build recipe name map for BOM validation (names from Sheet 1)
  const recipeNameMap = new Map<string, string>();
  for (const row of recipeHeaders) {
    const name = (row.name ?? '').trim().toLowerCase();
    if (name) recipeNameMap.set(name, 'pending');
  }

  // Validate recipe headers
  const validatedHeaders: ImportRow[] = [];
  for (let i = 0; i < recipeHeaders.length; i++) {
    const validated = await validateRecipeHeaderRow(recipeHeaders[i], i + 2, this.prisma);
    validatedHeaders.push(validated);
  }

  // Validate BOM lines
  const validatedBom: ImportRow[] = [];
  for (let i = 0; i < bomLines.length; i++) {
    const validated = await validateRecipeBomRow(bomLines[i], i + 2, this.prisma, recipeNameMap);
    validatedBom.push(validated);
  }

  // D-15: If recipe header is invalid/blocked, mark all its BOM lines as invalid too
  const invalidRecipeNames = new Set<string>();
  for (const row of validatedHeaders) {
    if (row.status === 'invalid' || row.status === 'blocked') {
      const name = ((row.raw.name ?? '') as string).trim().toLowerCase();
      if (name) invalidRecipeNames.add(name);
    }
  }
  for (const bomRow of validatedBom) {
    const recipeName = ((bomRow.raw.recipe_name ?? '') as string).trim().toLowerCase();
    if (invalidRecipeNames.has(recipeName) && bomRow.status !== 'invalid') {
      bomRow.status = 'invalid';
      bomRow.errors.push({ field: 'recipe_name', message: 'Parent recipe is invalid or blocked' });
    }
  }

  const config = IMPORT_TYPE_CONFIG[importType];
  const bomConfig = { columns: ['recipe_name', 'input_type', 'ingredient_name', 'quantity', 'unit', 'prep_notes'] };

  return {
    importType,
    rows: validatedHeaders,
    totalRows: validatedHeaders.length,
    validCount: validatedHeaders.filter(r => r.status === 'valid').length,
    invalidCount: validatedHeaders.filter(r => r.status === 'invalid').length,
    duplicateCount: validatedHeaders.filter(r => r.status === 'duplicate').length,
    blockedCount: validatedHeaders.filter(r => r.status === 'blocked').length,
    columns: config.columns,
    bomRows: validatedBom,
    bomColumns: bomConfig.columns,
    bomValidCount: validatedBom.filter(r => r.status === 'valid').length,
    bomInvalidCount: validatedBom.filter(r => r.status === 'invalid').length,
  } as any; // RecipeParseResult
}
```
