**Create CostCalculatorService** at `backend/src/recipes/cost-calculator.service.ts` (per RECIPE-06, D-19):

```typescript
@Injectable()
export class CostCalculatorService {
  constructor(private readonly prisma: PrismaService) {}

  async calculateRecipeCost(recipeId: string, visitedSet: Set<string> = new Set()): Promise<number | null> {
    if (visitedSet.has(recipeId)) return null; // cycle guard
    visitedSet.add(recipeId);

    const recipe = await this.prisma.recipe.findUnique({
      where: { id: recipeId },
      include: {
        RecipeLines: {
          include: {
            ingredient: {
              include: { VendorPrices: { orderBy: { effective_date: 'desc' }, take: 1 } },
            },
            source_recipe: true,
          },
        },
      },
    });

    if (!recipe || recipe.RecipeLines.length === 0) return null;

    let totalCost = 0;
    for (const line of recipe.RecipeLines) {
      if (line.input_type === 'ingredient') {
        const price = line.ingredient?.VendorPrices?.[0];
        if (!price) return null; // missing vendor price — cannot calculate
        const convertedQty = await convertUnit(Number(line.quantity), line.unit, price.unit, this.prisma);
        if (convertedQty === null) return null;
        totalCost += convertedQty * Number(price.price);
      } else if (line.input_type === 'recipe' && line.source_recipe_id) {
        const srcCost = await this.calculateRecipeCost(line.source_recipe_id, visitedSet);
        if (srcCost === null) return null;
        const srcYieldQty = Number(line.source_recipe!.yield_qty);
        if (srcYieldQty === 0) return null;
        const costPerUnit = srcCost / srcYieldQty;
        const convertedQty = await convertUnit(Number(line.quantity), line.unit, line.source_recipe!.yield_unit, this.prisma);
        if (convertedQty === null) return null;
        totalCost += costPerUnit * convertedQty;
      }
    }
    return totalCost;
  }

  async recalculateAndSave(recipeId: string): Promise<number | null> {
    const cost = await this.calculateRecipeCost(recipeId);
    await this.prisma.recipe.update({
      where: { id: recipeId },
      data: { computed_cost: cost },
    });
    return cost;
  }

  // Called by VendorsService when a VendorPrice is saved
  async recalculateForIngredient(ingredientId: string): Promise<void> {
    // Find all recipes that directly use this ingredient
    const directLines = await this.prisma.recipeLine.findMany({
      where: { ingredient_id: ingredientId },
      select: { recipe_id: true },
    });
    const directRecipeIds = [...new Set(directLines.map(l => l.recipe_id))];

    // Recalculate each direct recipe
    for (const recipeId of directRecipeIds) {
      await this.recalculateAndSave(recipeId);
    }

    // One level of propagation: find recipes that use any of the recalculated recipes as BOM input
    if (directRecipeIds.length > 0) {
      const parentLines = await this.prisma.recipeLine.findMany({
        where: { source_recipe_id: { in: directRecipeIds } },
        select: { recipe_id: true },
      });
      const parentRecipeIds = [...new Set(parentLines.map(l => l.recipe_id))];
      for (const recipeId of parentRecipeIds) {
        await this.recalculateAndSave(recipeId);
      }
    }
  }
}

const RECIPE_INCLUDE = {
  brand: { select: { id: true, name: true } },
  zone: { select: { id: true, name: true } },
  creator: { select: { id: true, name: true } },
  RecipeLines: {
    include: {

      source_recipe: {
        select: { id: true, name: true, yield_qty: true, yield_unit: true, computed_cost: true,
          RecipeLines: {
            include: {
              ingredient: { select: { id: true, name: true, base_unit: true, category: true } },
              source_recipe: { select: { id: true, name: true, yield_qty: true, yield_unit: true, computed_cost: true, RecipeLines: { include: { ingredient: { select: { id: true, name: true } }, source_recipe: { select: { id: true, name: true } } } } } },
            },
          },
        },
      },
    },
    orderBy: { sort_order: 'asc' as const },
  },
} as const;
```
Note: The nested include for dependency tree goes 3 levels deep. Deeper trees are navigated by clicking through to each recipe's detail page.

Methods:
- `findAll(filters: { brand_id?: string; status?: string; search?: string })` — findMany with where clause (brand_id, status, name contains search), include brand/zone/creator only (not full RecipeLines), orderBy name ASC
- `findOne(id: string)` — findUnique with RECIPE_INCLUDE, throw NotFoundException
- `create(dto: CreateRecipeDto, userId: string)` — Create recipe in $transaction: create recipe, then createMany RecipeLines from dto.bom_lines (if provided), then call costCalculatorService.recalculateAndSave(recipeId). Return with RECIPE_INCLUDE.
- `update(id: string, dto: UpdateRecipeDto, userId: string, isAdmin: boolean)` — findOne, check ownership (created_by === userId or isAdmin), update recipe fields, if dto.bom_lines provided then delete all existing RecipeLines + createMany new ones (BOM upsert pattern from research), recalculate cost. Status transition validation: if currently 'approved', cannot revert to 'draft' (throw BadRequestException).
- `remove(id: string)` — Check if used as source_recipe in other RecipeLines. If usage > 0, throw BadRequestException. Check if MenuItem references it. If referenced, throw BadRequestException. Otherwise delete (cascade deletes its own RecipeLines).
- `checkCycle(recipeId: string, sourceRecipeId: string)` — Before adding a BOM line with input_type='recipe', verify sourceRecipeId's transitive BOM does not already contain recipeId. Walk the source recipe's RecipeLines recursively with a visitedSet. Return true if cycle detected, false if safe.
